# ==============================================================================
# 1. CONFIGURE TOKEN AND DOWNLOAD DATASET ON SERVER (MARIMO / GPU SERVER)
# ==============================================================================
if [ -f "token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' token.txt | head -n 1)
elif [ -f "../token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' ../token.txt | head -n 1)
elif [ -f "/marimo/token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' /marimo/token.txt | head -n 1)
elif [ -f "/marimo/Capstone_2026_Fish_Feeding_Intensity/token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' /marimo/Capstone_2026_Fish_Feeding_Intensity/token.txt | head -n 1)
fi

if [ -z "$HF_TOKEN" ]; then
    echo "⚠️ HF_TOKEN not found! Please create token.txt (echo 'hf_xxx' > /marimo/token.txt) or run: export HF_TOKEN='hf_xxx'"
fi

git config --global credential.helper store
if [ -n "$HF_TOKEN" ]; then
    hf auth login --token "$HF_TOKEN" --add-to-git-credential || huggingface-cli login --token "$HF_TOKEN" || true
fi
sudo apt update && sudo apt install git-lfs -y
git lfs install
git config --global lfs.concurrenttransfers 64
cd /marimo && nohup git clone https://huggingface.co/datasets/manhmitcf/Fish_Feeding_Intensity_Dataset > /marimo/clone.log 2>&1 &
tail -n +1 -f /marimo/clone.log

# ==============================================================================
# 2. REORGANIZE DATASET DIRECTORY STRUCTURE
# ==============================================================================
mv /marimo/Fish_Feeding_Intensity_Dataset/audio/audio/* /marimo/Fish_Feeding_Intensity_Dataset/audio/
mv /marimo/Fish_Feeding_Intensity_Dataset/video/video/* /marimo/Fish_Feeding_Intensity_Dataset/video/
rm -rf /marimo/Fish_Feeding_Intensity_Dataset/audio/audio
rm -rf /marimo/Fish_Feeding_Intensity_Dataset/video/video
find /marimo/Fish_Feeding_Intensity_Dataset -type f -iname "*.mp4" | wc -l
find /marimo/Fish_Feeding_Intensity_Dataset -type f -iname "*.wav" | wc -l

# ==============================================================================
# 3. TRAINING: Video Temporal Mode Ablation (Branch: ablations/video_temporal_mode)
#    (6-ch ConvNeXt-Nano: composite_ln vs sum vs abs_diff)
#    Artifact Upload Repo: manhmitcf/video_temporal_mode (Branch: main)
# ==============================================================================
if [ ! -d "/marimo/Capstone_2026_Fish_Feeding_Intensity" ]; then
    git clone https://github.com/manhmitcf/Capstone_2026.git /marimo/Capstone_2026_Fish_Feeding_Intensity
fi
cd /marimo/Capstone_2026_Fish_Feeding_Intensity
git fetch origin
git checkout ablations/video_temporal_mode
git pull origin ablations/video_temporal_mode
cd /marimo/Capstone_2026_Fish_Feeding_Intensity/U_FFIA27K_multimodal
pip install --upgrade pip
pip install -r requirements.txt
python main.py --dry-run
pkill -f "python main.py" || true
nohup python main.py > main.log 2>&1 &
tail -n +1 -f main.log

# ==============================================================================
# 4. RESUME TRAINING FROM LAST CHECKPOINT (IF INTERRUPTED)
# ==============================================================================
pkill -f "python main.py" || true
nohup python main.py --resume > main.log 2>&1 &
tail -n +1 -f main.log

# ==============================================================================
# 5. MONITOR TRAINING PROGRESS
# ==============================================================================
tail -n +1 -f /marimo/Capstone_2026_Fish_Feeding_Intensity/U_FFIA27K_multimodal/main.log
